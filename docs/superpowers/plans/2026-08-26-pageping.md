# PagePing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build PagePing — a free, hosted, open-source universal webpage watcher (paste any URL, get emailed when it changes / keyword appears / price drops), deployed on `*.convex.site`.

**Architecture:** Vite+React SPA served by the Convex static-hosting component. All logic in Convex: pure change-detection engine (shared client/server file), Firecrawl scrape actions, hourly cron re-checks, AgentMail OTP auth + alerts + track-by-email webhook. Content hashes every check; full snapshots stored only on change (cap 20).

**Tech Stack:** TypeScript, Vite, React 19, Tailwind v4, Convex, `@firecrawl/firecrawl-convex`, `@agentmail/convex`, `@convex-dev/static-hosting`, `diff`, `tsx` test runner (node:test).

**Spec:** `docs/superpowers/specs/2026-08-26-pageping-design.md`

```ts
import { diffLines } from "diff";

export type Condition = "any-change" | "keyword" | "price-below";
export const ALERT_MIN_GAP_MS = 6 * 60 * 60 * 1000;

export function normalizeContent(text: string): string {
  return text.split("\n").map((l) => l.trim()).filter(Boolean).join("\n");
}

export async function sha256Hex(text: string): Promise<string> {
  const bytes = new TextEncoder().encode(text);
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest)).map((b) => b.toString(16).padStart(2, "0")).join("");
}

const CURRENCIES: Record<string, string> = { "₹": "₹", "$": "$", "€": "€", "£": "£" };

// ponytail: first price match wins; per-page heuristics only if accuracy complaints
export function extractPrice(text: string): { price: number; currency: string } | null {
  const re = /([₹$€£])\s?([\d][\d,]*(?:\.\d{1,2})?)/;
  const m = text.match(re);
  if (!m) return null;
  const price = parseFloat(m[2].replace(/,/g, ""));
  if (!Number.isFinite(price) || price <= 0 || price > 10_000_000) return null;
  return { price, currency: CURRENCIES[m[1]] };
}

export interface EvalInput {
  condition: Condition;
  keyword?: string;
  targetPrice?: number;
  prevHash?: string | null;
  prevHadKeyword: boolean;
  nextHash: string;
  nextText: string;
  nextPrice: number | null;
  lastAlertedAt?: number | null;
  lastAlertedPrice?: number | null;
  now: number;
}
export interface EvalOutput { changed: boolean; alert: "change" | "keyword" | "price" | null }

export function evaluate(input: EvalInput): EvalOutput {
  const changed = input.prevHash == null || input.prevHash !== input.nextHash;
  let alert: EvalOutput["alert"] = null;
  if (input.condition === "any-change") {
    alert = changed ? "change" : null;
  } else if (input.condition === "keyword" && input.keyword) {
    const has = input.nextText.toLowerCase().includes(input.keyword.toLowerCase());
    alert = has && !input.prevHadKeyword ? "keyword" : null;
  } else if (input.condition === "price-below" && input.targetPrice != null) {
    const p = input.nextPrice;
    if (p != null && p <= input.targetPrice &&
        (input.lastAlertedPrice == null || p < input.lastAlertedPrice)) {
      alert = "price";
    }
  }
  const last = input.lastAlertedAt ?? 0;
  if (input.now - last < ALERT_MIN_GAP_MS) alert = null;
  return { changed, alert };
}

export interface DiffRow { type: " " | "+" | "-"; line: string }

export function buildDiff(oldText: string, newText: string): DiffRow[] {
  const parts = diffLines(oldText, newText);
  const rows: DiffRow[] = [];
  for (const part of parts) {
    const type: DiffRow["type"] = part.added ? "+" : part.removed ? "-" : " ";
    for (const line of part.value.replace(/\n$/, "").split("\n")) {
      if (type !== " ") rows.push({ type, line });   // context rows skipped: diffs stay focused
      else if (rows.filter((r) => r.type === " ").length < 6) rows.push({ type, line });
    }
  }
  return rows;
}
```

```ts
import { ConvexError } from "convex/values";
import type { DatabaseReader, DatabaseWriter } from "./_generated/server";
import type { Doc, Id } from "./_generated/dataModel";

export async function requireUser(db: DatabaseReader, sessionId: string): Promise<Doc<"users">> {
  const s = await db.query("sessions").withIndex("by_token", (q) => q.eq("token", sessionId)).unique();
  if (!s || s.expiresAt < Date.now()) throw new ConvexError("UNAUTHORIZED");
  const user = await db.get(s.userId);
  if (!user) throw new ConvexError("UNAUTHORIZED");
  return user;
}

export async function ensureUser(db: DatabaseWriter, email: string): Promise<Id<"users">> {
  const existing = await db.query("users").withIndex("by_email", (q) => q.eq("email", email)).unique();
  if (existing) return existing._id;
  return db.insert("users", { email, createdAt: Date.now() });
}
```

```ts
import { action, mutation } from "./_generated/server";
import { internalMutation, internalQuery } from "./_generated/server";
import { v } from "convex/values";
import { ConvexError } from "convex/values";
import { components, internal } from "./_generated/api";
import { AgentMail } from "@agentmail/convex";
import { sha256Hex } from "../src/lib/engine";
import { ensureUser } from "./lib";

const agentmail = new AgentMail(components.agentmail);

async function getOrCreateInbox(ctx: any): Promise<{ inboxId: string; address: string }> {
  const row = await ctx.runQuery(internal.auth.getConfigRow, {});
  if (row) return { inboxId: row.value.split("|")[0], address: row.value.split("|")[1] };
  // README-gated (Task 4 Step 2): confirm createInbox signature; adapt here if needed.
  const inbox = await agentmail.createInbox(ctx);
  await ctx.runMutation(internal.auth.saveConfig, { value: `${inbox.inboxId}|${inbox.email}` });
  return { inboxId: inbox.inboxId, address: inbox.email };
}

export const getConfigRow = internalQuery({
  args: {},
  handler: async (ctx) =>
    await ctx.db.query("config").withIndex("by_key", (q) => q.eq("key", "agentmail_inbox")).unique(),
});

export const saveConfig = internalMutation({
  args: { value: v.string() },
  handler: async (ctx, { value }) => {
    const row = await ctx.db.query("config").withIndex("by_key", (q) => q.eq("key", "agentmail_inbox")).unique();
    if (!row) await ctx.db.insert("config", { key: "agentmail_inbox", value });
  },
});

export const requestOtp = action({
  args: { email: v.string() },
  handler: async (ctx, { email }) => {
    const clean = email.trim().toLowerCase();
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(clean)) throw new ConvexError("INVALID_EMAIL");
    const code = String(Math.floor(100000 + Math.random() * 900000));
    await ctx.runMutation(internal.auth.storeOtp, { email: clean, codeHash: await sha256Hex(`${clean}:${code}`) });
    if (process.env.AUTH_DEBUG === "true") return { debugCode: code }; // ponytail: dev-only escape hatch; delete AUTH_DEBUG before prod
    const inbox = await getOrCreateInbox(ctx);
    await agentmail.sendMessage(ctx, inbox.inboxId, {
      to: clean,
      subject: `Your PagePing code: ${code}`,
      text: `Your PagePing login code is ${code}. It expires in 10 minutes.`,
    });
    return {};
  },
});

export const storeOtp = internalMutation({
  args: { email: v.string(), codeHash: v.string() },
  handler: async (ctx, { email, codeHash }) => {
    const expiresAt = Date.now() + 10 * 60 * 1000;
    const row = await ctx.db.query("otpCodes").withIndex("by_email", (q) => q.eq("email", email)).unique();
    if (row) await ctx.db.patch(row._id, { codeHash, expiresAt });
    else await ctx.db.insert("otpCodes", { email, codeHash, expiresAt });
  },
});

export const verifyOtp = mutation({
  args: { email: v.string(), code: v.string() },
  handler: async (ctx, { email, code }) => {
    const clean = email.trim().toLowerCase();
    const row = await ctx.db.query("otpCodes").withIndex("by_email", (q) => q.eq("email", clean)).unique();
    if (!row || row.expiresAt < Date.now()) throw new ConvexError("OTP_EXPIRED");
    if (row.codeHash !== (await sha256Hex(`${clean}:${code.trim()}`))) throw new ConvexError("WRONG_CODE");
    await ctx.db.delete(row._id);
    const userId = await ensureUser(ctx.db, clean);
    const token = Array.from(crypto.getRandomValues(new Uint8Array(32))).map((b) => b.toString(16).padStart(2, "0")).join("");
    await ctx.db.insert("sessions", { userId, token, expiresAt: Date.now() + 30 * 24 * 3600 * 1000 });
    return { sessionId: token };
  },
});
```

```ts
export const createWatch = mutation({
  args: {
    sessionId: v.string(),
    url: v.string(),
    condition: v.union(v.literal("any-change"), v.literal("keyword"), v.literal("price-below")),
    keyword: v.optional(v.string()),
    targetPrice: v.optional(v.number()),
  },
  handler: async (ctx, { sessionId, url, condition, keyword, targetPrice }) => {
    const user = await requireUser(ctx.db, sessionId);
    let parsed: URL;
    try { parsed = new URL(url); } catch { throw new ConvexError("BAD_URL"); }
    if (parsed.protocol !== "http:" && parsed.protocol !== "https:") throw new ConvexError("BAD_URL");
    if (condition === "keyword" && !keyword?.trim()) throw new ConvexError("KEYWORD_REQUIRED");
    if (condition === "price-below" && (targetPrice == null || targetPrice <= 0)) throw new ConvexError("TARGET_REQUIRED");

    const mine = await ctx.db.query("watches").withIndex("by_owner", (q) => q.eq("ownerEmail", user.email)).collect();
    if (mine.length >= MAX_WATCHES) throw new ConvexError("LIMIT_REACHED");
    const dup = mine.find((w) => w.url === url && w.status === "active");
    if (dup) return { watchId: dup._id, existed: true };

    const watchId = await ctx.db.insert("watches", {
      ownerEmail: user.email,
      url,
      title: "",
      condition,
      keyword: condition === "keyword" ? keyword!.trim() : undefined,
      targetPrice: condition === "price-below" ? targetPrice : undefined,
      status: "active",
      deadNotified: false,
      failureCount: 0,
      lastCheckedAt: 0,
      createdAt: Date.now(),
    });
    await ctx.scheduler.runAfter(0, internal.check.checkWatch, { watchId });
    return { watchId, existed: false };
  },
});
```

```ts
import { internalAction, internalMutation, internalQuery } from "./_generated/server";
import { internal } from "./_generated/api";
import { v } from "convex/values";
import { FirecrawlClient } from "@firecrawl/firecrawl-convex";
import { components } from "./_generated/api";
import { normalizeContent, sha256Hex, extractPrice, evaluate } from "../src/lib/engine";

const firecrawl = new FirecrawlClient(components.firecrawl);

export const loadWatch = internalQuery({
  args: { watchId: v.id("watches") },
  handler: async (ctx, { watchId }) => await ctx.db.get(watchId),
});

export const checkWatch = internalAction({
  args: { watchId: v.id("watches") },
  handler: async (ctx, { watchId }) => {
    const w = await ctx.runQuery(internal.check.loadWatch, { watchId });
    if (!w || w.status !== "active") return;
    try {
      // ponytail: maxAge 0 always fresh; enable caching if Firecrawl credits burn fast
      const res = await firecrawl.scrape(ctx, w.url, {
        formats: ["markdown"], onlyMainContent: true, maxAge: 0,
      });
      const markdown = res.markdown ?? "";
      const found = extractPrice(markdown);
      await ctx.runMutation(internal.check.recordCheck, {
        watchId,
        markdown,
        title: res.metadata?.title ?? w.title,
        price: found?.price ?? null,
        currency: found?.currency ?? null,
      });
    } catch {
      await ctx.runMutation(internal.check.recordFailure, { watchId });
    }
  },
});

export const recordCheck = internalMutation({
  args: {
    watchId: v.id("watches"),
    markdown: v.string(),
    title: v.string(),
    price: v.union(v.number(), v.null()),
    currency: v.union(v.string(), v.null()),
  },
  handler: async (ctx, { watchId, markdown, title, price, currency }) => {
    const w = await ctx.db.get(watchId);
    if (!w) return;
    const now = Date.now();
    const normalized = normalizeContent(markdown);
    const hash = await sha256Hex(normalized);

    const prevSnap = await ctx.db.query("snapshots")
      .withIndex("by_watch_time", (q) => q.eq("watchId", watchId))
      .order("desc").first();
    const prevHadKeyword = !!prevSnap && !!w.keyword &&
      prevSnap.markdown.toLowerCase().includes(w.keyword.toLowerCase());

    const out = evaluate({
      condition: w.condition, keyword: w.keyword, targetPrice: w.targetPrice,
      prevHash: w.lastHash ?? null, prevHadKeyword,
      nextHash: hash, nextText: normalized, nextPrice: price,
      lastAlertedAt: w.lastAlertedAt ?? null, lastAlertedPrice: w.lastAlertedPrice ?? null,
      now,
    });

    if (out.changed) {
      await ctx.db.insert("snapshots", {
        watchId, checkedAt: now, contentHash: hash, markdown: normalized,
        ...(price != null ? { price } : {}),
      });
      const snaps = await ctx.db.query("snapshots")
        .withIndex("by_watch_time", (q) => q.eq("watchId", watchId)).collect();
      if (snaps.length > 20) {
        const oldest = [...snaps].sort((a, b) => a.checkedAt - b.checkedAt);
        for (const s of oldest.slice(0, snaps.length - 20)) await ctx.db.delete(s._id);
      }
    }

    await ctx.db.patch(watchId, {
      lastCheckedAt: now,
      lastHash: hash,
      failureCount: 0,
      ...(title && !w.title ? { title } : {}),
      ...(price != null ? { currentPrice: price } : {}),
      ...(currency != null ? { currency } : {}),
    });

    if (out.alert) {
      await ctx.scheduler.runAfter(0, internal.alerts.sendAlert, {
        watchId, kind: out.alert, price,
      });
    }
  },
});

export const recordFailure = internalMutation({
  args: { watchId: v.id("watches") },
  handler: async (ctx, { watchId }) => {
    const w = await ctx.db.get(watchId);
    if (!w) return;
    const failureCount = w.failureCount + 1;
    await ctx.db.patch(watchId, { failureCount, lastCheckedAt: Date.now() });
    if (failureCount >= 5 && w.status === "active") {
      await ctx.db.patch(watchId, { status: "dead" });
      if (!w.deadNotified) {
        await ctx.db.patch(watchId, { deadNotified: true });
        await ctx.scheduler.runAfter(0, internal.alerts.sendDeadNotice, { watchId });
      }
    }
  },
});
```

```tsx
import { useQuery } from "@convex/react";
import { api } from "../../convex/_generated/api";
import { Id } from "../../convex/_generated/dataModel";
import DiffView from "../components/DiffView";
import AddWatchForm from "../components/AddWatchForm";

function ago(ts: number) {
  const m = Math.max(1, Math.round((Date.now() - ts) / 60000));
  return m < 60 ? `${m}m ago` : `${Math.round(m / 60)}h ago`;
}

export default function WatchPage({ id }: { id: string }) {
  const data = useQuery(api.watches.getPublic, { watchId: id as Id<"watches"> });

  if (data === undefined) return <div className="mx-auto max-w-2xl p-8 text-zinc-500">Loading…</div>;
  if (data === null) return <div className="mx-auto max-w-2xl p-8">Watch not found.</div>;

  const { watch, snapshots } = data;
  const latest = snapshots[0];
  const previous = snapshots[1];

  return (
    <div className="mx-auto max-w-2xl px-4 py-8">
      <header className="mb-6 flex items-center justify-between">
        <h1 className="cursor-pointer text-xl font-bold" onClick={() => window.history.pushState({}, "", "/")}>PagePing</h1>
        <span className={`rounded-full px-2 py-1 text-xs ${watch.status === "active" ? "bg-emerald-950 text-emerald-400" : "bg-red-950 text-red-400"}`}>
          {watch.status}
        </span>
      </header>

      <section className="rounded-xl border border-zinc-800 bg-zinc-900 p-4">
        <h2 className="font-semibold">{watch.title || watch.url}</h2>
        <a href={watch.url} target="_blank" rel="noreferrer" className="text-xs text-zinc-500 underline">{watch.url}</a>
        <div className="mt-3 flex flex-wrap gap-4 text-sm">
          {watch.currentPrice != null && (
            <span>Price now: <b className="text-emerald-400">{watch.currency}{watch.currentPrice}</b></span>
          )}
          {watch.targetPrice != null && <span className="text-zinc-400">Target: {watch.currency}{watch.targetPrice}</span>}
          {watch.keyword && <span className="text-zinc-400">Watching for: “{watch.keyword}”</span>}
          <span className="text-zinc-500">{watch.lastCheckedAt ? `Checked ${ago(watch.lastCheckedAt)}` : "First check pending…"}</span>
        </div>
      </section>

      <section className="mt-6">
        <h3 className="mb-2 text-sm font-semibold uppercase tracking-wide text-zinc-400">Latest change</h3>
        {latest && previous ? (
          <>
            <p className="mb-2 text-xs text-zinc-500">{ago(previous.checkedAt)} → {ago(latest.checkedAt)}</p>
            <DiffView oldText={previous.markdown} newText={latest.markdown} />
          </>
        ) : (
          <p className="text-sm text-zinc-500">
            {latest ? "Only one snapshot so far — no change detected yet." : "No snapshots yet."}
          </p>
        )}
      </section>

      <section className="mt-6">
        <h3 className="mb-2 text-sm font-semibold uppercase tracking-wide text-zinc-400">History ({snapshots.length} recent changes)</h3>
        <ol className="space-y-1 text-sm text-zinc-400">
          {snapshots.map((s) => (
            <li key={s.contentHash + s.checkedAt} className="flex justify-between border-b border-zinc-800 py-1">
              <span>Change detected</span><span className="text-zinc-500">{ago(s.checkedAt)}</span>
            </li>
          ))}
        </ol>
      </section>

      <section className="mt-8 rounded-xl border border-dashed border-zinc-700 p-4">
        <h3 className="mb-2 font-semibold">Want updates like this?</h3>
        <AddWatchForm presetUrl={watch.url} />
      </section>
    </div>
  );
}
```
